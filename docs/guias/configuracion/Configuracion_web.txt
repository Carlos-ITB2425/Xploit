const isLoggedIn = <?php echo isset($_SESSION['user_id']) ? 'true' : 'false'; ?>;

function checkAuthAndGo(url) {
    if (isLoggedIn) {
        location.href = url;
    } else {
        showModal('login');
    }
}

const meses = ['Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio', 'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'];
const daySel = document.getElementById('rday');
const monthSel = document.getElementById('rmonth');
const yearSel = document.getElementById('ryear');

for (let i = 2026; i >= 1920; i--) {
    yearSel.innerHTML += `<option value="${i}">${i}</option>`;
}

meses.forEach((m, index) => {
    monthSel.innerHTML += `<option value="${(index + 1).toString().padStart(2, '0')}">${m}</option>`;
});

function updateDays() {
    let y = yearSel.value || 2026;
    let m = monthSel.value || 1;
    let daysInMonth = new Date(y, m, 0).getDate();
    let currentDay = daySel.value;
    daySel.innerHTML = '<option value="">DÍA</option>';
    for (let i = 1; i <= daysInMonth; i++) {
        let val = i.toString().padStart(2, '0');
        daySel.innerHTML += `<option value="${val}" ${currentDay === val ? 'selected' : ''}>${i}</option>`;
    }
}

monthSel.addEventListener('change', updateDays);
yearSel.addEventListener('change', updateDays);
updateDays();

function showModal(id) {
    document.getElementById('modal' + id.charAt(0).toUpperCase() + id.slice(1)).style.display = 'flex';
}

window.onclick = e => {
    if (e.target.className === 'modal') e.target.style.display = 'none';
};

async function auth(type) {
    const fd = new FormData();

    if (type === 'login') {
        fd.append('action', 'login');
        fd.append('identifier', document.getElementById('luser').value);
        fd.append('password', document.getElementById('lpass').value);
    } else {
        let d = daySel.value, m = monthSel.value, y = yearSel.value;
        daySel.classList.remove('error');
        monthSel.classList.remove('error');
        yearSel.classList.remove('error');

        if (!d || !m || !y) {
            if (!d) daySel.classList.add('error');
            if (!m) monthSel.classList.add('error');
            if (!y) yearSel.classList.add('error');
            return;
        }

        fd.append('action', 'register');
        fd.append('nombre', document.getElementById('rnom').value);
        fd.append('apellido', document.getElementById('rape').value);
        fd.append('fechanac', `${y}-${m}-${d}`);
        fd.append('username', document.getElementById('ruser').value);
        fd.append('email', document.getElementById('rmail').value);
        fd.append('password', document.getElementById('rpass').value);
    }

    const res = await fetch('auth.php', { method: 'POST', body: fd });
    const data = await res.json();
    if (data.success) location.reload();
    else alert(data.message);
}

    function buildTargetUrl() {
        const protocol = window.location.protocol + '//';
        const host = window.location.hostname;
        return protocol + host + ':6969';
    }

    function openConnectionModal() {
        const modal = document.getElementById('connectionModal');
        const finalUrl = buildTargetUrl();

        document.getElementById('urlFlag1').textContent = finalUrl;
        document.getElementById('urlFlag2').textContent = finalUrl;
        document.getElementById('urlFlag3').textContent = finalUrl;

        if (modal) {
            modal.classList.add('active');
        }
    }

    function closeConnectionModal() {
        const modal = document.getElementById('connectionModal');
        if (modal) {
            modal.classList.remove('active');
        }
    }

    function openTarget(event) {
        event.preventDefault();
        window.open(buildTargetUrl(), '_blank');
    }

    window.addEventListener('click', function(event) {
        const modal = document.getElementById('connectionModal');
        if (event.target === modal) {
            closeConnectionModal();
        }
    });

    window.addEventListener('keydown', function(event) {
        if (event.key === 'Escape') {
            closeConnectionModal();
        }
    });
</script>

    function openSshModal() {
        const modal = document.getElementById('sshModal');
        if (modal) {
            modal.classList.add('active');
        }
    }

    function closeSshModal() {
        const modal = document.getElementById('sshModal');
        if (modal) {
            modal.classList.remove('active');
        }
    }

    function copyToClipboard() {
        const command = document.getElementById('sshCommand');
        const btn = document.getElementById('copyBtn');

        if (!command || !btn) return;

        const textToCopy = command.textContent.trim();
        const originalText = 'COPIAR';

        if (navigator.clipboard && window.isSecureContext) {
            navigator.clipboard.writeText(textToCopy).then(function () {
                btn.innerText = '¡COPIADO!';
                btn.style.background = 'var(--neon-green)';
                btn.style.color = '#000';

                setTimeout(function () {
                    btn.innerText = originalText;
                    btn.style.background = 'var(--neon-purple)';
                    btn.style.color = '#000';
                }, 2000);
            }).catch(function () {
                fallbackCopy(textToCopy, btn, originalText);
            });
        } else {
            fallbackCopy(textToCopy, btn, originalText);
        }
    }

    function fallbackCopy(text, btn, originalText) {
        const tempInput = document.createElement('textarea');
        tempInput.value = text;
        tempInput.style.position = 'fixed';
        tempInput.style.left = '-9999px';
        document.body.appendChild(tempInput);
        tempInput.focus();
        tempInput.select();

        try {
            document.execCommand('copy');
            btn.innerText = '¡COPIADO!';
            btn.style.background = 'var(--neon-green)';
            btn.style.color = '#000';

            setTimeout(function () {
                btn.innerText = originalText;
                btn.style.background = 'var(--neon-purple)';
                btn.style.color = '#000';
            }, 2000);
        } catch (err) {
            alert('Error al copiar al portapapeles.');
        }

        document.body.removeChild(tempInput);
    }

    window.addEventListener('click', function(event) {
        const modal = document.getElementById('sshModal');
        if (event.target === modal) {
            closeSshModal();
        }
    });

    window.addEventListener('keydown', function(event) {
        if (event.key === 'Escape') {
            closeSshModal();
        }
    });
</script>

===== auth.php =====
<?php
session_start();
require_once 'db.php';

header('Content-Type: application/json; charset=utf-8');
ini_set('display_errors', '0');
error_reporting(E_ALL);

try {
    $action = $_POST['action'] ?? '';

    if ($action === 'login') {
        $identifier = trim($_POST['identifier'] ?? '');
        $password = $_POST['password'] ?? '';

        $stmt = $pdo->prepare("SELECT id, username, passwordhash FROM usuarios WHERE username = ? OR email = ? LIMIT 1");
        $stmt->execute([$identifier, $identifier]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user && password_verify($password, $user['passwordhash'])) {
            $_SESSION['user_id'] = $user['id'];
            $_SESSION['username'] = $user['username'];
            echo json_encode(['success' => true, 'message' => 'Login correcto.']);
        } else {
            echo json_encode(['success' => false, 'message' => 'Credenciales inválidas.']);
        }
        exit;
    }

    if ($action === 'register') {
        $nombre = trim($_POST['nombre'] ?? '');
        $apellido = trim($_POST['apellido'] ?? '');
        $fechanac = trim($_POST['fechanac'] ?? '');
        $username = trim($_POST['username'] ?? '');
        $email = trim($_POST['email'] ?? '');
        $password = $_POST['password'] ?? '';

        $hash = password_hash($password, PASSWORD_BCRYPT);

        $stmt = $pdo->prepare("INSERT INTO usuarios (nombre, apellido, fechanacimiento, username, email, passwordhash, puntos) VALUES (?, ?, ?, ?, ?, ?, 0)");
        $stmt->execute([$nombre, $apellido, $fechanac, $username, $email, $hash]);

        echo json_encode(['success' => true, 'message' => 'Usuario registrado correctamente.']);
        exit;
    }

    echo json_encode(['success' => false, 'message' => 'Acción no válida.']);
} catch (Throwable $e) {
    echo json_encode(['success' => false, 'message' => 'Error interno.']);
}
